cmake_minimum_required(VERSION 3.15)
project(jacobsrays LANGUAGES CXX)
add_library(jacobsrays INTERFACE)
target_include_directories(jacobsrays INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(jacobsrays INTERFACE cxx_std_11)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_core tests/test_core.cpp)
    target_link_libraries(test_core PRIVATE jacobsrays)
    add_test(NAME core COMMAND test_core)
endif()
option(JACOBSRAYS_MATLAB "Build MATLAB MEX gateway" OFF)
if(JACOBSRAYS_MATLAB)
    find_package(Matlab REQUIRED COMPONENTS MEX_COMPILER)
    matlab_add_mex(NAME jacobsrays_mex SRC matlab/jacobsrays_mex.cpp)
    target_link_libraries(jacobsrays_mex PRIVATE jacobsrays)
endif()
